## 3) Remplacer la fonction `switchTab`

Chercher :

```js
function switchTab(id){document.body.dataset.theme=id;document.querySelectorAll('main').forEach(m=>m.classList.remove('activePage'));q(id).classList.add('activePage');document.querySelectorAll('.tabs button').forEach(b=>b.classList.toggle('active',b.dataset.tab===id));render()}
```

Remplacer par :

```js
function switchTab(id){
  document.body.dataset.theme=id;
  document.querySelectorAll('main').forEach(m=>m.classList.remove('activePage'));
  q(id).classList.add('activePage');
  document.querySelectorAll('.tabs button').forEach(b=>{
    const active=b.dataset.tab===id;
    b.classList.toggle('active',active);
    if(active){
      b.scrollIntoView({behavior:'smooth',inline:'center',block:'nearest'});
    }
  });
  render();
}
```

## 4) Ajouter un choix de catégorie pour “Surprise culture”

Dans la section `Culture`, juste avant :

```html
<div class="compactList" id="cultureList"></div>
```

Ajouter :

```html
<div class="sortLine">
  <select id="cultureSurpriseFilter">
    <option>Tous</option>
    <option>Film</option>
    <option>Série</option>
    <option>Disney</option>
    <option>Marvel</option>
    <option>Classiques français</option>
    <option>Classiques internationaux</option>
    <option>Cinéma moderne</option>
    <option>Blockbusters</option>
    <option>Comédies</option>
  </select>
  <button class="soft" id="cultureSurpriseBtn">Surprise dans cette catégorie</button>
</div>
```

Puis remplacer la fonction `cultureSurprise()` par :

```js
function cultureSurprise(){
  const select=q('cultureSurpriseFilter');
  const filter=select?select.value:'Tous';
  let pool=state.culture.filter(x=>x.status!=='Vu');
  if(filter!=='Tous'){
    pool=pool.filter(x=>x.type===filter||x.collection===filter);
  }
  if(!pool.length){
    setFocus('🎉 Rien à tirer','Aucun contenu à voir dans cette catégorie.');
    switchTab('culture');
    return;
  }
  let x=pool[Math.floor(Math.random()*pool.length)];
  setFocus('🍿 '+x.title,(x.collection||'Culture')+(x.platform?' · '+x.platform:''));
  switchTab('culture');
}
```

Enfin, dans les listeners, ajouter :

```js
if(q('cultureSurpriseBtn'))q('cultureSurpriseBtn').addEventListener('click',cultureSurprise);
```
